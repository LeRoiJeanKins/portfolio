To automate contests, copy and paste this script into your chrome extension tamperMonkey :

You must enable user scripts, access to file URLs, and be using Google Chrome. 

// ==UserScript==
// @name         Instant Gaming Auto Giveaway
// @author       LeRoiJeanKins
// @description  participer automatiquements aux concours ig
// @version      1.0
// @match        *://www.instant-gaming.com/*
// @match        https://game-giveaway-list.netlify.app
// @match        http://localhost:*/*
// @run-at       document-idle
// @grant        GM_registerMenuCommand
// @grant GM_openInTab
// @license      MIT
// ==/UserScript==

// Ce script est une version modifiée du script original de enzomtp,
// adapté et amélioré par LeRoiJeanKins
// lien vers le github originel  https://github.com/enzomtpYT/InstantGamingGiveawayList

(function() {
  "use strict";


    if (!document.location.href.endsWith('?igr=LeRoiJeanKins') && document.location.href.startsWith('https://www.instant-gaming.com/')) {
      if(document.location.href.includes("?igr=")){
        let index
        let finDuLien
        index = document.location.href.indexOf("?")
        finDuLien = document.location.href.slice(index)
        document.location.href = document.location.href.replace(finDuLien,'?igr=LeRoiJeanKins')

      }
      else{
        document.location.href = document.location.href+'?igr=LeRoiJeanKins'
      }
    }

  function openInNewTab(url) {
       window.open(url, '_blank');

  }

  function participate(){
    // Get the participate button element.
    const participate = document.querySelector("button.button.validate");
    // Click on participate if it exists.
    if (participate !== null) {
      participate.click();
      location.reload();
    }
  }

  function socials(){
    // Get all the socials
    const socials = document.querySelectorAll("a.button.reward")
    // Click on each socials
    socials.forEach((e) => e.click())
  }

  function giveawayList(){
    // Get the giveaway list
    const giveawayList = document.querySelectorAll("a.giveaway")
    // Click on the giveaway list
    giveawayList.forEach((e) => e.click())
  }

  function openall() {
      const links = document.querySelectorAll("#resultat > a");
      links.forEach((link, index) => {
          setTimeout(() => {
              GM_openInTab(link.href, { active: false }); 
          }, index * 2500);
      });
  }
  if (document.location.hostname === "www.instant-gaming.com") {
          setTimeout(() => {
              participate();
              socials();
               window.close();
          }, 1500);
  }



  // Register the menu command
  GM_registerMenuCommand("Participate", participate);
  GM_registerMenuCommand("Socials", socials);
  GM_registerMenuCommand("Giveaway List", giveawayList);
  GM_registerMenuCommand("Open ALL Links", openall);
  participate();
  socials();

})();

How the site works: 


This site is a personal project that lists Instant Gaming giveaways I’ve found.

To participate, simply click on each image. It will open the giveaway link in a new tab where you can join directly.

The giveaway colors indicate your click status:
🟩 Green: you've already clicked this link before.
🟦 Blue: you clicked it during this session.
🟥 Red: you've never clicked it before.

If you want to enter all giveaways automatically, go to the "Automation" section and follow the steps.
Once the script is installed via the TamperMonkey extension, a button labeled "Open Links" will appear, allowing you to automatically enter all giveaways (make sure to allow pop-ups on this site).
